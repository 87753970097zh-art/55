let currentVariant = 0;
let currentQuestion = 0;
let score = 0;
let answered = false;


function startQuiz(variant, button) {

    currentVariant = variant;
    currentQuestion = 0;
    score = 0;

    document.querySelectorAll(".variant-btn")
        .forEach(btn => btn.classList.remove("active"));

    button.classList.add("active");

    document.getElementById("quiz").style.display = "block";
    document.getElementById("result").style.display = "none";

    showQuestion();
}


function showQuestion() {

    answered = false;

    const data = tests[currentVariant][currentQuestion];

    document.getElementById("questionNumber").textContent =
        `${currentQuestion + 1} / ${tests[currentVariant].length}`;

    document.getElementById("question").textContent = data.q;

    const answers = document.getElementById("answers");
    answers.innerHTML = "";

    document.getElementById("feedback").textContent = "";

    document.getElementById("nextBtn").style.display = "none";

    const progress =
        ((currentQuestion) / tests[currentVariant].length) * 100;

    document.getElementById("progressBar").style.width =
        progress + "%";


    data.a.forEach((answer, index) => {

        const button = document.createElement("button");

        button.className = "answer";

        button.textContent =
            String.fromCharCode(65 + index) + ") " + answer;

        button.onclick = () =>
            checkAnswer(button, index);

        answers.appendChild(button);
    });
}


function checkAnswer(button, selected) {

    if (answered) return;

    answered = true;

    const data = tests[currentVariant][currentQuestion];

    const allButtons =
        document.querySelectorAll(".answer");

    allButtons.forEach(btn =>
        btn.classList.add("disabled")
    );


    if (selected === data.correct) {

        button.classList.add("correct");

        score++;

        document.getElementById("feedback").innerHTML =
            "🎉 Дұрыс жауап! Жарайсың! ⭐";

        createConfetti();

    } else {

        button.classList.add("wrong");

        allButtons[data.correct]
            .classList.add("correct");

        document.getElementById("feedback").innerHTML =
            "💡 Қате жауап. Дұрыс жауапты қара!";

    }

    document.getElementById("nextBtn").style.display =
        "inline-block";
}


function nextQuestion() {

    currentQuestion++;

    if (currentQuestion >= tests[currentVariant].length) {

        showResult();

    } else {

        showQuestion();

    }
}


function showResult() {

    document.getElementById("quiz").style.display = "none";

    document.getElementById("result").style.display = "block";

    document.getElementById("score").textContent =
        `${score} / 10`;

    let stars = "";

    if (score === 10) {
        stars = "⭐⭐⭐⭐⭐";
    }
    else if (score >= 8) {
        stars = "⭐⭐⭐⭐";
    }
    else if (score >= 6) {
        stars = "⭐⭐⭐";
    }
    else if (score >= 4) {
        stars = "⭐⭐";
    }
    else {
        stars = "⭐";
    }

    document.getElementById("stars").textContent = stars;


    let text = "";

    if (score === 10) {
        text = "🏆 Керемет! Барлық сұраққа дұрыс жауап бердің!";
        createConfetti();
    }
    else if (score >= 8) {
        text = "👏 Өте жақсы нәтиже! Сен информатиканы жақсы меңгеріп келесің!";
    }
    else if (score >= 6) {
        text = "👍 Жақсы! Тағы да қайталап жаттықсаң, нәтижең одан да жоғары болады.";
    }
    else {
        text = "📚 Сабақ тақырыптарын тағы бір қайталап көр!";
    }

    document.getElementById("resultText").textContent = text;
}


function restartQuiz() {

    currentQuestion = 0;
    score = 0;

    document.getElementById("result").style.display = "none";
    document.getElementById("quiz").style.display = "block";

    showQuestion();
}


function createConfetti() {

    const symbols = ["🎉","⭐","✨","🎈","🌟","💻"];

    for (let i = 0; i < 25; i++) {

        const item = document.createElement("div");

        item.className = "confetti";

        item.textContent =
            symbols[Math.floor(Math.random() * symbols.length)];

        item.style.left =
            Math.random() * 100 + "vw";

        item.style.animationDuration =
            (1.5 + Math.random() * 2) + "s";

        document.body.appendChild(item);

        setTimeout(() => {
            item.remove();
        }, 4000);
    }
}
